const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId(); function formatDateIndo(dateStr) { if (!dateStr) return ""; const months = [ "Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember" ]; const d = new Date(dateStr); const day = d.getDate(); const month = months[d.getMonth()]; const year = d.getFullYear(); return `${day} ${month} ${year}`; } function normalizePhone(phoneStr) { if (!phoneStr || phoneStr === "-") return "-"; let cleaned = phoneStr.toString().replace(/\D/g, ''); if (cleaned.startsWith('0')) cleaned = '62' + cleaned.slice(1); if (cleaned.startsWith('620')) cleaned = '62' + cleaned.slice(3); if (cleaned.startsWith('62628')) cleaned = cleaned.slice(2); return cleaned; } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName("Orders"); // 1. CREATE if (data.action === "add_order") { const orderId = "ORD-" + Utilities.formatDate(new Date(), "GMT+7", "yyyyMMdd-HHmmss"); const cleanPhone = normalizePhone(data.phone); sheet.appendRow([ orderId, cleanPhone, data.name, data.item, Number(data.price), data.orderDate, data.deadline, "Pending", data.type || "Buy" ]); const formattedDate = formatDateIndo(data.orderDate); const suggestedName = `IJS ${data.type} ${data.name} ${data.item} ${formattedDate}`; const contactSheet = ss.getSheetByName("Contacts"); if (contactSheet) { contactSheet.appendRow([cleanPhone, data.name, formattedDate, suggestedName]); } return responseJSON({ status: "success", suggestedName: suggestedName, orderId: orderId }); } // 2. UPDATE STATUS if (data.action === "update_status") { const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (values[i][0] === data.orderId) { sheet.getRange(i + 1, 8).setValue(data.newStatus); return responseJSON({ status: "success" }); } } return responseJSON({ status: "error", message: "ID tidak ditemukan" }); } // 3. EDIT ORDER if (data.action === "edit_order") { const values = sheet.getDataRange().getValues(); const cleanPhone = normalizePhone(data.phone); for (let i = 1; i < values.length; i++) { if (values[i][0] === data.orderId) { sheet.getRange(i + 1, 2).setValue(cleanPhone); sheet.getRange(i + 1, 3).setValue(data.name); sheet.getRange(i + 1, 4).setValue(data.item); sheet.getRange(i + 1, 5).setValue(Number(data.price)); sheet.getRange(i + 1, 6).setValue(data.orderDate); sheet.getRange(i + 1, 7).setValue(data.deadline); sheet.getRange(i + 1, 9).setValue(data.type); return responseJSON({ status: "success" }); } } return responseJSON({ status: "error", message: "ID tidak ditemukan" }); } // 4. DELETE ORDER if (data.action === "delete_order") { const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (values[i][0] === data.orderId) { sheet.deleteRow(i + 1); return responseJSON({ status: "success" }); } } return responseJSON({ status: "error", message: "ID tidak ditemukan" }); } } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } function doGet(e) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName("Orders"); if (!sheet) { return responseJSON({ status: "success", data: [] }); } const data = sheet.getDataRange().getValues(); data.shift(); const orders = data.map(row => ({ id: row[0], phone: normalizePhone(row[1]), name: row[2], item: row[3], price: row[4], orderDate: row[5] ? Utilities.formatDate(new Date(row[5]), "GMT+7", "yyyy-MM-dd") : "", deadline: row[6] ? Utilities.formatDate(new Date(row[6]), "GMT+7", "yyyy-MM-dd") : "", status: row[7] || "Pending", type: row[8] || "Buy" })); return responseJSON({ status: "success", data: orders }); } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
